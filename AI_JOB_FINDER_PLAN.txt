  ```python
  import bm25s, os
  INDEX_DIR = r"D:\Ai job finder\data\bm25_index"

  def build_bm25_index(corpus: list[dict]):
      texts = [doc["text"] for doc in corpus]
      tokens = bm25s.tokenize(texts, stopwords="en")
      retriever = bm25s.BM25()
      retriever.index(tokens)
      os.makedirs(INDEX_DIR, exist_ok=True)
      retriever.save(INDEX_DIR, corpus=corpus)
      return retriever

  def query_bm25(query_str: str, top_k: int = 25):
      retriever = bm25s.BM25.load(INDEX_DIR, mmap=True)
      query_tokens = bm25s.tokenize(query_str)
      docs, scores = retriever.retrieve(query_tokens, k=top_k)
      return docs[0], scores[0]
  ```

  ```python
  import subprocess, os

  def render_typst_resume(typst_content: str, output_pdf_path: str):
      temp_typ = output_pdf_path.replace(".pdf", ".typ")
      with open(temp_typ, "w", encoding="utf-8") as f:
          f.write(typst_content)
      try:
          cmd = ["typst", "compile", temp_typ, output_pdf_path]
          subprocess.run(cmd, check=True, capture_output=True, text=True)
      finally:
          if os.path.exists(temp_typ):
              os.remove(temp_typ)
  ```
